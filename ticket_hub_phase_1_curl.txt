SECTION 11: FULL CUSTOMER BOOKING JOURNEY (ONE-LINER AUTOMATED SCRIPT)
================================================================================
Copy and paste this entire block in your terminal to run the complete end-to-end flow:

```bash
GATEWAY="http://localhost:8080"
# For testing from another laptop on LAN, uncomment this line:
# GATEWAY="http://10.10.21.86:8080"

echo "1. Authenticating as Customer..."
TOKEN=$(curl -s -X POST "$GATEWAY/auth/login" \
  -H "Content-Type: application/json" \
  -d '{"username":"customer","password":"customer123"}' | grep -o '"token":"[^"]*' | cut -d'"' -f4)
echo "   Token: ${TOKEN:0:30}..."

echo "2. Finding available seat for Show #100..."
SEAT_ID=$(curl -s "$GATEWAY/api/shows/100/seats" -H "Authorization: Bearer $TOKEN" | \
  python3 -c "import sys, json; s=[x['seatId'] for x in json.load(sys.stdin).get('data', []) if x['status']=='AVAILABLE']; print(s[0] if s else '')")
echo "   Selected Available Seat: $SEAT_ID"

echo "3. Locking seat for 10 minutes..."
curl -s -X POST "$GATEWAY/api/shows/100/seats/lock" \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d "{\"showId\": 100, \"userId\": 101, \"seatIds\": [$SEAT_ID]}" > /dev/null
echo "   Seat $SEAT_ID locked successfully."

echo "4. Creating Booking..."
IDEM_KEY="run-$(date +%s)-$RANDOM"
BKG_RESP=$(curl -s -X POST "$GATEWAY/api/bookings" \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -H "Idempotency-Key: $IDEM_KEY" \
  -d "{\"showId\": 100, \"seatIds\": [$SEAT_ID], \"userId\": 101, \"customerEmail\": \"customer@tickethub.com\", \"customerName\": \"TicketHub Customer\"}")
BKG_ID=$(echo "$BKG_RESP" | grep -o '"bookingId":[0-9]*' | cut -d':' -f2)
echo "   Booking created with ID: $BKG_ID"

echo "5. Initiating Payment..."
PAY_RESP=$(curl -s -X POST "$GATEWAY/api/payments" \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d "{\"bookingId\": $BKG_ID, \"amount\": 250.00, \"currency\": \"INR\", \"paymentMethod\": \"UPI\", \"idempotencyKey\": \"pay-$IDEM_KEY\"}")
PAY_ID=$(echo "$PAY_RESP" | grep -o '"paymentId":[0-9]*' | cut -d':' -f2)
ORDER_ID=$(echo "$PAY_RESP" | grep -o '"razorpayOrderId":"[^"]*' | cut -d'"' -f4)
echo "   Payment initiated with ID: $PAY_ID, Order: $ORDER_ID"

echo "6. Confirming Payment..."
curl -s -X POST "$GATEWAY/api/payments/$PAY_ID/confirm" \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d "{\"razorpayOrderId\": \"$ORDER_ID\", \"razorpayPaymentId\": \"pay_sim_$(date +%s)\"}" > /dev/null
echo "   Payment confirmed successfully."

echo "7. Retrieving Confirmed Ticket..."
TICKET=$(curl -s "$GATEWAY/api/bookings/$BKG_ID/ticket" -H "Authorization: Bearer $TOKEN")
echo "   Ticket Details:"
echo "$TICKET" | python3 -m json.tool

echo "8. Customer Journey Completed 100% Successfully!"
```
